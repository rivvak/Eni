    "stream", "stop", "presence_penalty", "frequency_penalty"
}

@app.post("/v1/chat/completions")
async def chat_completions(request: Request):
    """
    Maximum hardening chat completions endpoint.
    - Handles streaming and non-streaming
    - Applies ENI/VEIL persona injection + pure injection defense
    - Blocks ALL system message injection from clients
    - Forces model to FORCED_MODEL (glm-5.1 or glm-5.2)
    """
    
    try:
        body = await request.json()
    except Exception as e:
        return JSONResponse(
            status_code=400,
            content={"error": f"Invalid JSON: {str(e)}"}
        )
    
    # Extract and sanitize messages (MAXIMUM HARDENING)
    original_messages = body.get("messages", [])
    hardened_messages = sanitize_messages(original_messages)
    
    # Force our model override
    body["model"] = FORCED_MODEL
    body["messages"] = hardened_messages
    
    # Filter to allowed keys only
    filtered_body = {k: v for k, v in  body.items() if k in ALLOWED_CHAT_KEYS}
    
    # Determine if streaming
    is_streaming = filtered_body.get("stream", False)
    
    # Headers for uncensored.com
    headers = {
        "Content-Type": "application/json",
        "X-api-key": UNCENSORED_API_KEY,
    }
    
    # Use async httpx client
    async with httpx.AsyncClient(timeout=TIMEOUT) as client:
        
        if is_streaming:
            # ═══════════════════════════════════════════════════════
            # STREAMING MODE
            # ═══════════════════════════════════════════════════════
            
            try:
                upstream = await client.post(
                    f"{UNCENSORED_API_BASE}/chat/completions",
                    json=filtered_body,
                    headers=headers,
                )
                
                async def stream_proxy():
                    """Stream SSE chunks directly from uncensored.com"""
                    try:
                        async for chunk in upstream.aiter_raw():
                            yield chunk
                    except Exception as e:
                        error_msg = f"data: {json.dumps({'error': str(e)})}\n\n"
                        yield error_msg.encode('utf-8')
                
                return StreamingResponse(
                    stream_proxy(),
                    media_type="text/event-stream",
                    headers={
                        "Cache-Control": "no-cache",
                        "Connection": "keep-alive",
                        "X-Accel-Buffering": "no",
                    }
                )
                
            except Exception as e:
                return JSONResponse(
                    status_code=500,
                    content={"error": f"Streaming request failed: {str(e)}"}
                )
        
        else:
            # ═══════════════════════════════════════════════════════
            # NON-STREAMING MODE
            # ═══════════════════════════════════════════════════════
            
            try:
                resp = await client.post(
                    f"{UNCENSORED_API_BASE}/chat/completions",
                    json=filtered_body,
                    headers=headers,
                )
                
                # Try to parse JSON response
                try:
                    data = resp.json()
                    return JSONResponse(
                        status_code=resp.status_code,
                        content=data
                    )
                except ValueError:
                    # If not JSON, expose raw text for debugging
                    return JSONResponse(
                        status_code=resp.status_code,
                        content={
                            "error": "Non-JSON response from uncensored.com",
                            "status_code": resp.status_code,
                            "raw": resp.text[:2000],  # Limit raw text to 2000 chars
                        }
                    )
                    
            except Exception as e:
                return JSONResponse(
                    status_code=500,
                    content={"error": f"Non-streaming request failed: {str(e)}"}
                )
